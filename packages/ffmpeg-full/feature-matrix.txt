# ffmpeg-full 目标功能矩阵（机器可读声明）。
#
# scripts/verify-ffmpeg-features.sh 在每次构建后对真实产物逐条断言；缺失即
# 构建失败（configure 对 --enable-lib* 是尽力而为，依赖探测失败会静默降级，
# 只有对产物本身的断言能拦住这种静默丢失）。
#
# 格式：<class> <token>
#   flag      ffmpeg -buildconf 的 configuration 里必须出现 --enable-<token>
#   hwaccel   ffmpeg -hwaccels 列表里的独立一行 <token>
#   encoder   ffmpeg -encoders 表中的 <token>（第二字段）
#   decoder   ffmpeg -decoders 表中的 <token>
#   filter    ffmpeg -filters 表中的 <token>
#   needed    产物 ELF DT_NEEDED 里以 <token> 开头的库（readelf -d）
#   dlopen    产物 strings 引用的运行时 dlopen 库名（ffmpeg 9 起 vapoursynth
#             等集成无 DT_NEEDED,DT_NEEDED 断言结构性失败）
#
# 增删条目 = 修改本文件；CI 首跑若暴露假红（token 与 9.0.2 实际命名不符），
# 修正这里而不是放宽脚本。

# --- configure 开关（overlay 注入 + AUR 原生目标功能） ---
# libnpp 已在 ffmpeg 9.0.2 上游移除("libnpp has been removed and enabling
# it does nothing",CI run 43921345810),不再断言。
flag libplacebo
flag libsvtav1
flag libsvthevc
flag libsvtjpegxs
flag libsvtvp9
flag libopenvino
flag libopencv
flag vapoursynth
flag whisper
flag vulkan
flag opencl
flag cuda
flag cuvid
flag nvenc
flag nvdec
flag ffnvcodec
flag libdrm
flag vaapi
flag vdpau
flag libvpl
flag libx264
flag libx265
flag libdav1d
flag libaom
flag libfdk-aac
flag gpl
flag version3
flag nonfree
flag shared

# --- 硬件加速方法 ---
hwaccel cuda
# cuvid hwaccel 类型在 ffmpeg 9.0.2 的 -hwaccels 输出已不存在(cuvid 解码器
# 由 decoder h264_cuvid/hevc_cuvid 断言兜底)。
hwaccel drm
hwaccel vaapi
hwaccel vdpau
hwaccel vulkan
hwaccel qsv
hwaccel opencl

# --- 编码器 ---
encoder h264_nvenc
encoder hevc_nvenc
encoder libsvtav1
# SVT 补丁与 rav1e 的注册名带下划线/前缀(9.0.2 Enabled encoders 实测)。
encoder libsvt_hevc
encoder libsvt_vp9
encoder librav1e
encoder libsvtjpegxs
encoder libx264
encoder libx265
encoder libopenh264
encoder libfdk_aac
encoder libmp3lame
encoder aac

# --- 解码器 ---
decoder h264
decoder hevc
decoder av1
decoder libdav1d
decoder h264_cuvid
decoder hevc_cuvid

# --- 滤镜 ---
filter libplacebo
filter scale_vulkan
filter overlay_vulkan
filter transpose_vulkan
# scale_npp/transpose_npp/overlay_npp 随 libnpp 上游移除,不再断言。
filter whisper
filter zscale

# --- 直接链接的库（DT_NEEDED 前缀） ---
needed libplacebo
needed libopenvino
dlopen libvapoursynth-script
